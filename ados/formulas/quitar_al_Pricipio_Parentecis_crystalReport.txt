Local StringVar glosa;
Local NumberVar posCierre;

If IsNull({ado.Prod.Glosa}) Then
""

Else
{
	glosa := {ado.Prod.Glosa};
	posCierre := InStr(glosa, ")");

//Si empieza con "(" y encuentra el cierre ")", quita solo ese bloque inicial
	If Left(glosa, 1) = "(" And posCierre > 0 Then
		glosa := Mid(glosa, posCierre +1);
	Trim(glosa)
)
